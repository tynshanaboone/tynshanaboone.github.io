# every API call goes through this; the key comes from .Renviron and goes in a header
govinfo <- function(path) {
  request("https://api.govinfo.gov") |>
    req_url_path_append(path) |>
    req_headers(`X-Api-Key` = Sys.getenv("GOVINFO_API_KEY"), .redact = "X-Api-Key") |>
    req_retry(max_tries = 3)
}

# save each API response once, then read the saved copy
cached <- function(file, f) {
  path <- file.path(dir, file)
  if (!file.exists(path)) saveRDS(f(), path)
  readRDS(path)
}

      offsetMark = offset_mark,
      sorts      = list(list(field = "publishdate", sortOrder = "DESC"))

2d. Paging

search_all <- function(query) {
  out  <- list()
  mark <- "*"
  repeat {
    page <- search_page(query, mark)
    if (length(page$results) == 0) break
    out  <- c(out, page$results)
    mark <- page$offsetMark
    if (is.null(mark)) break
  }
  bind_rows(map(out, \(r) r[c("packageId", "title", "dateIssued", "lastModified")]))
}

3. Download the documents

3a. The ten most recent, two ways

Part A is the download_govfiles_pdf() function from govtdata01.R, run on the pdfLink column of my export. My copy of the script has no Part B, so I wrote it with httr2 from section 2: it takes the ten newest from the API search in 2b and fetches each from /packages/{packageId}/pdf. Both wait 1 to 3 seconds between files.

api_top10    <- api_all |> slice_head(n = 10) |> mutate(rank = row_number())
export_top10 <- export  |> slice_head(n = 10) |> mutate(rank = row_number())

log_file <- file.path(dir, "download_log.csv")

# Part A: the function from govtdata01.R
save_dir <- paste0(pdf_dir, "/")
download_govfiles_pdf <- function(url, id) {
  tryCatch({
    destfile <- paste0(save_dir, "govfiles_", id, ".pdf")
    download.file(url, destfile = destfile, mode = "wb") # Binary files
    Sys.sleep(runif(1, 1, 3))
    return(paste("Successfully downloaded:", url))
  },
  error = function(e) {
    return(paste("Failed to download:", url))
  })
}

# Part B: one package at a time from the API
download_api_pdf <- function(id) {
  destfile <- file.path(pdf_dir, paste0("api_", id, ".pdf"))
  resp <- govinfo(paste0("packages/", id, "/pdf")) |>
    req_error(is_error = \(r) FALSE) |>
    req_perform(path = destfile)
  Sys.sleep(runif(1, 1, 3))
  paste("HTTP", resp_status(resp))
}

# one log row per file, with the time it finished
log_row <- function(method, attempt, rank, id, url, file, result) {
  tibble(method, attempt, rank, packageId = id, url, file, result,
         bytes = file.size(file), md5 = unname(tools::md5sum(file)),
         finished = format(Sys.time(), "%Y-%m-%d %H:%M:%S %Z", tz = "America/Chicago"))
}

run_a <- function(rows, attempt) {
  pmap(list(rows$rank, rows$packageId, rows$pdfLink), \(rank, id, url) {
    result <- download_govfiles_pdf(url, id)
    log_row("A: export + download.file()", attempt, rank, id, url,
            paste0(save_dir, "govfiles_", id, ".pdf"), result)
  }) |> list_rbind()
}

if (!file.exists(log_file)) {
  log_a1 <- run_a(export_top10, 1)

  # retry the failures with a longer timeout than R's default of 60 seconds
  failed <- export_top10 |> filter(packageId %in% log_a1$packageId[str_starts(log_a1$result, "Failed")])
  old <- options(timeout = 600)
  log_a2 <- run_a(failed, 2)
  options(old)

  log_b <- pmap(list(api_top10$rank, api_top10$packageId), \(rank, id) {
    result <- download_api_pdf(id)
    log_row("B: API /packages/{id}/pdf", 1, rank, id,
            paste0("https://api.govinfo.gov/packages/", id, "/pdf"),
            file.path(pdf_dir, paste0("api_", id, ".pdf")), result)
  }) |> list_rbind()

  bind_rows(log_a1, log_a2, log_b) |> write_csv(log_file)
}

dl_log <- read_csv(log_file, col_types = cols(.default = "c")) |>
  mutate(across(c(attempt, rank, bytes), as.numeric))

Part B got all 10 files on the first try. Part A failed on 2 large PDFs (CHRG-119shrg62398 and CHRG-119shrg61656). R stops a download after 60 seconds by default, and the script’s tryCatch() returned “Failed” but left the partial file on disk (7,427,944 and 6,303,248 bytes). Without the log, those would look like real PDFs. The retry with options(timeout = 600) got both. Which files fail depends on download speed; in an earlier test run a different large file failed. httr2 has no time limit by default, so Part B was not affected.

The two columns of senators end up on the same lines. Pages carry page numbers (“(II)” in the front matter, a number at the top of body pages), and words are split across lines with hyphens (“agree-”). To turn ten PDFs into a table or a corpus I would need to:

  1. Strip page numbers, headers and line-break hyphens, and rejoin paragraphs that cross pages.
  2. Split the front matter (roster, contents) from the hearing itself.
  3. Find each speaker turn (“The Chairman.”, “Senator Shaheen.”, “Mr. DiNanno.”) and give it a speaker name, party and role.
  4. Separate the spoken hearing from inserted material such as prepared statements and answers to questions for the record.
  5. Join each turn to the hearing’s metadata (date, title, S. Hrg. number) to get one row per speaker turn.

parse_mods <- function(x, id) {
  doc <- xml2::read_xml(x) |> xml2::xml_ns_strip()
  n   <- \(path) length(xml2::xml_find_all(doc, path))
  tibble(
    packageId = id,
    held      = xml2::xml_text(xml2::xml_find_first(doc, ".//extension/heldDate")),
    members   = n(".//extension/congMember"),
    D         = n(".//extension/congMember[@party='D']"),
    R         = n(".//extension/congMember[@party='R']"),
    witnesses = n(".//extension/witness"),
    first_witness = xml2::xml_text(xml2::xml_find_first(doc, ".//extension/witness")) |> coalesce("none listed")
  )
}
